        function speak(text) {

            let msg = new SpeechSynthesisUtterance(text);

            msg.lang = 'en-US';

            window.speechSynthesis.speak(msg);

        }


        function checkAnswer(inputId, correctAns) {

            let userAns = document.getElementById(inputId).value.trim();

            let resId = inputId.replace('ans', 'res');

            if (userAns === correctAns) {

                document.getElementById(resId).innerHTML = "✅ 答啱咗啦！好叻女！";

                document.getElementById(resId).style.color = "green";

            } else {

                document.getElementById(resId).innerHTML = "❌ 差少少，再試吓啦！(答案係: " + correctAns + ")";

                document.getElementById(resId).style.color = "red";

            }

        }
